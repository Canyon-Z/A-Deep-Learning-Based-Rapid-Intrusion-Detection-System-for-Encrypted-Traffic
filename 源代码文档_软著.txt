   15 | class PureBiLSTM(nn.Module):
   16 |     def __init__(self, input_size=28, hidden_dim=64, num_layers=2, num_classes=2):
   17 |         super(PureBiLSTM, self).__init__()
   18 |         self.bilstm = nn.LSTM(input_size=input_size, hidden_size=hidden_dim, 
   19 |                               num_layers=num_layers, batch_first=True, bidirectional=True, 
   20 |                               dropout=0.5 if num_layers > 1 else 0)
   21 |         self.dropout = nn.Dropout(0.5)
   22 |         # BiLSTM output will be concatenated (hidden_dim * 2)
   23 |         # Sequence length after flatten is image width (28)
   24 |         self.fc = nn.Linear(hidden_dim * 2, num_classes)
   25 |         
   26 |     def forward(self, x):
   27 |         # x shape: [batch, 1, 28, 28] -> remove channel dim -> [batch, 28, 28]
   28 |         x = x.squeeze(1)
   29 |         # lstm returns: output, (h_n, c_n)
   30 |         output, _ = self.bilstm(x)
   31 |         # use the last sequence state output[:, -1, :] 
   32 |         last_state = output[:, -1, :]
   33 |         last_state = self.dropout(last_state)
   34 |         out = self.fc(last_state)
   35 |         return out
